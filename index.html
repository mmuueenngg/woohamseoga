<script>

// 한국어 모스 부호

// 우: ㅇ + ㅜ

// 함: ㅎ + ㅏ + ㅁ

// 서: ㅅ + ㅓ

// 가: ㄱ + ㅏ


const syllables = [

  { name: "우", jamo: ["-.-", "...."] },

  { name: "함", jamo: [".---", ".", "--"] },

  { name: "서", jamo: ["--.", "-"] },

  { name: "가", jamo: [".-..", "."] }

];


// 신호와 침묵을 시간 단위로 구성

const segments = [];


syllables.forEach((syllable, sIndex) => {

  syllable.jamo.forEach((code, jIndex) => {

    const symbols = code.split("");


    symbols.forEach((symbol, i) => {

      segments.push({

        type: symbol,

        units: symbol === "." ? 1 : 3,

        word: syllable.name,

        index: sIndex

      });


      // 한 자모 안에서 신호 사이의 간격

      if (i < symbols.length - 1) {

        segments.push({ type: "gap", units: 1 });

      }

    });


    // 자모 사이의 간격

    if (jIndex < syllable.jamo.length - 1) {

      segments.push({ type: "gap", units: 3 });

    }

  });


  // 음절 사이의 간격

  if (sIndex < syllables.length - 1) {

    segments.push({ type: "gap", units: 5 });

  }

});


const totalUnits = segments.reduce(

  (sum, segment) => sum + segment.units, 0

);


const music = document.getElementById("music");

const currentWord = document.getElementById("currentWord");


const lines = syllables.map((syllable, index) => {

  const line = document.getElementById("line" + index);


  syllable.jamo.join("").split("").forEach(symbol => {

    const dot = document.createElement("div");

    dot.className = "signal " + (symbol === "-" ? "dash" : "dot");

    line.appendChild(dot);

  });


  return line;

});


function updateAnimation() {

  if (!music.duration || !Number.isFinite(music.duration)) return;


  const progress = music.currentTime / music.duration;

  const position = progress * totalUnits;


  let elapsed = 0;

  let activeSegment = null;


  for (const segment of segments) {

    if (

      position >= elapsed &&

      position < elapsed + segment.units

    ) {

      activeSegment = segment;

      break;

    }

    elapsed += segment.units;

  }


  // 모든 신호를 우선 어둡게

  lines.forEach(line => {

    [...line.children].forEach(dot => {

      dot.classList.remove("active");

    });

  });


  if (activeSegment && activeSegment.type !== "gap") {

    const line = lines[activeSegment.index];


    // 현재 음절에서 해당 기호를 찾아 활성화

    const signals = [...line.children];

    const signalIndex = signals.findIndex(dot => {

      return dot.dataset.used !== "true" &&

        dot.classList.contains(

          activeSegment.type === "-" ? "dash" : "dot"

        );

    });


    // 현재 진행 중인 신호 위치 계산

    let count = 0;

    const currentCode = syllables[activeSegment.index].jamo

      .join("");


    for (let i = 0; i < activeSegment.index; i++) {

      count += 1;

    }


    const precedingSignals = segments

      .slice(0, segments.indexOf(activeSegment))

      .filter(segment =>

        segment.index === activeSegment.index &&

        segment.type !== "gap"

      ).length;


    const dot = signals[precedingSignals];

    if (dot) dot.classList.add("active");


    currentWord.textContent =

      "SIGNAL : " + activeSegment.word;

  } else {

    currentWord.textContent = " ";

  }

}


let animationFrame;


function animate() {

  updateAnimation();

  if (!music.paused && !music.ended) {

    animationFrame = requestAnimationFrame(animate);

  }

}


music.addEventListener("play", () => {

  cancelAnimationFrame(animationFrame);

  animate();

});


music.addEventListener("pause", () => {

  cancelAnimationFrame(animationFrame);

  updateAnimation();

});


music.addEventListener("timeupdate", updateAnimation);

music.addEventListener("seeked", updateAnimation);


music.addEventListener("ended", () => {

  cancelAnimationFrame(animationFrame);

  updateAnimation();

  currentWord.textContent = "END";

});
